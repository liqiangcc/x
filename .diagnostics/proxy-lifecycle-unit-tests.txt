            process.stdout.write(text);
          }
        });
        child.stderr.on("data", (chunk) => {
          const text = chunk.toString();
          stderr += text;
          if (options.echoStderr !== false) {
            process.stderr.write(text);
          }
        });
        child.on("error", reject);
        child.on("close", (code) => {
          const result = { stdout, stderr, streamed: true };
          if (code === 0) {

          error.code = code ?? 1;

          error.streamed = true;
          reject(error);
        });
      });
    }

    function selectRouterProbeRegions(report) {
      const results = Array.isArray(report?.results) ? report.results : [];
      return results
        .filter((result) => result.engine === "aws-router" && result.ok && result.region)
        .sort((left, right) => {
          const leftDuration = probeDuration(left);
          const rightDuration = probeDuration(right);
          if (Number.isFinite(leftDuration) && Number.isFinite(rightDuration) && leftDuration !== rightDuration) {
            return leftDuration - rightDuration;
          }
          if (Number.isFinite(leftDuration)) {
            return -1;
          }
          if (Number.isFinite(rightDuration)) {
            return 1;
          }
          return String(left.region).localeCompare(String(right.region));
        })
        .map((result) => result.region)
        .filter((region, index, regions) => regions.indexOf(region) === index);
    }

    async function inspectKlineCodes(period, codes) {
      const files = uniqueCodes(codes).flatMap((code) => [
        shardedKlinePath(period, code),
        legacyKlinePath(period, code),
      ]);
      const issues = [];
      let cursor = 0;
      let totalFiles = 0;
    
      await Promise.all(Array.from({ length: Math.min(16, files.length) }, async () => {
        while (cursor < files.length) {
          const file = files[cursor++];
          try {
            await fs.access(file);
          } catch (error) {
            if (error.code === "ENOENT") continue;
            throw error;
          }
          totalFiles += 1;
          const issue = await inspectFile(file);
          if (issue) issues.push(issue);
        }
      }));
    
      return {
        target_path: path.join(DEFAULT_KLINE_DIR, period),
        total_files: totalFiles,
        empty_count: issues.length,
        issue_count: issues.length,
        status: issues.length > 0 ? "failed" : "ok",
        issues,
      };
    }
    
    async function cleanupInvalidKlineFiles(period, codes = null) {
      let summary;

        if (Array.isArray(codes)) {
          summary = await inspectKlineCodes(period, codes);
        } else {
          summary = await validateKlinePath({ targetPath: DEFAULT_KLINE_DIR, period });
        }
      } catch (error) {
        if (error?.code !== "ENOENT") {
          throw error;
        }
        return {
          target_path: path.join(DEFAULT_KLINE_DIR, period),
          total_files: 0,
          issue_count: 0,
          status: "ok",
          issues: [],
          removed_codes: [],
        };
      }
      const issues = Array.isArray(summary.issues) ? summary.issues : [];
      const removedCodes = [];
    
      for (const issue of issues) {
        if (!issue?.file) {
          continue;
        }
        await fs.rm(issue.file, { force: true });
        if (issue.code) {
          removedCodes.push(String(issue.code));
        }
      }
    
      return {
        ...summary,
        removed_codes: uniqueCodes(removedCodes),
      };
    }

    function latestKlineDate(payload) {
      const klines = getKlines(payload);
      if (!Array.isArray(klines.value) || klines.value.length === 0) {
        return null;
      }
    
      for (let index = klines.value.length - 1; index >= 0; index -= 1) {
        const row = klines.value[index];
        if (typeof row !== "string") {
          continue;
        }
        const date = row.split(",")[0];
        if (/^\d{4}-\d{2}-\d{2}$/.test(date)) {
          return date;
        }
      }
    
      return null;
    }

    async function findStaleKlineCodes({ expectedLatestDate, freshnessCodes, period }) {
      if (!["daily", "yearly"].includes(period) || !expectedLatestDate || !freshnessCodes || freshnessCodes.size === 0) {
        return [];
      }
    
      const staleCodes = [];
      for (const code of freshnessCodes) {
        let filePath = shardedKlinePath(period, code);
        try {
          await fs.access(filePath);
        } catch {
          filePath = legacyKlinePath(period, code);
        }
    
        let payload;
        try {
          payload = await readJson(filePath);
        } catch {
          staleCodes.push(code);
          continue;
        }
    
        const latestDate = latestKlineDate(payload);
        if (!latestDate || latestDate < expectedLatestDate) {
          staleCodes.push(code);
        }
      }
    
      return uniqueCodes(staleCodes);
    }

        });

      progress.last_batch_source = selection.source;

    async function hasCompletePoolSnapshot(date, outputDir) {
      const poolDir = path.join(ROOT, outputDir, date);
      for (const pool of ["dt", "qs", "zb", "zt"]) {
        try {
          const payload = JSON.parse(await fs.readFile(path.join(poolDir, `${pool}.json`), "utf8"));
          if (String(payload?.data?.qdate ?? "") !== date) {
            return false;
          }
          if (!Array.isArray(payload?.data?.pool)) {
            return false;
          }
        } catch {
          return false;
        }
      }
      return true;
    }
    async function readExistingJson(filePath) {
      try {
        return JSON.parse(await fs.readFile(filePath, "utf8"));
      } catch {
        return null;
      }
    }

      const subcommand = argv[0];
      if (subcommand === "pool") {
        await commandProxyPool(argv.slice(1));

    function parseDurationMs(value) {
      const match = String(value ?? "").trim().match(/^(\d+)(s|m|h)$/i);
      if (!match || Number(match[1]) < 1) {
        throw new Error(`Invalid duration: ${value}. Use values such as 30s, 30m, or 1h.`);
      }
      const multiplier = { s: 1000, m: 60_000, h: 3_600_000 }[match[2].toLowerCase()];
      return Number(match[1]) * multiplier;
    }
    
    function parsePositiveOption(value, name, fallback) {
      if (value === undefined || value === null || value === "") {
        return fallback;
      }
      const number = Number(value);
      if (!Number.isInteger(number) || number < 1) {
        throw new Error(`${name} must be a positive integer.`);
      }
      return number;
    }

        try {
          let startIndex = 0;
          do {

            rounds.push({ ...report, passed: report.available_count > 0, started_at: new Date().toISOString() });

    }
